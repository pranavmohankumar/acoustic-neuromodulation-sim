import numpy as np

def simulate_fdtd_wave_2d(grid_size=(100, 100), steps=200):
    p = np.zeros(grid_size)
    p_prev = np.zeros(grid_size)
    c = np.ones(grid_size) * 1500.0 # Speed of sound m/s
    c[40:60, :] = 2800.0 # Skull bone barrier
    
    dt = 1e-6
    dx = 1e-3
    
    for t in range(steps):
        laplacian = (np.pad(p[1:, :], ((0,1),(0,0))) + np.pad(p[:-1, :], ((1,0),(0,0))) +
                     np.pad(p[:, 1:], ((0,0),(0,1))) + np.pad(p[:, :-1], ((0,0),(1,0))) - 4*p) / (dx**2)
        p_next = 2*p - p_prev + (c**2) * (dt**2) * laplacian
        p_prev = p.copy()
        p = p_next.copy()
        p[10, 50] = np.sin(2 * np.pi * 500e3 * t * dt) # Source
    return p
